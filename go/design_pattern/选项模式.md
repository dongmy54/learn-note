## 选项模式
主要是一种,通过传递多个选项函数，解决配置多样化的问题;
主要利用指针去改变指针值方式实现

### 举例
`service.go`选项包

```go
package server

import (
	"time"
)

type Service struct {
	host    string
	port    int64
	timeout time.Duration
	maxConn int64
}

// 选项函数类型
// 传入service的指针 才能改变原来的值
type Option func(svr *Service) *Service

// 初始化接收多个选项函数
func NewService(opts ...Option) *Service {
	svr := &Service{}
	// 在内部遍历选项函数，挨着执行
	for _, opt := range opts {
		// 给选项函数传参数指针,进入内部修改参数
		svr = opt(svr)
	}
	return svr
}

func WithHost(host string) Option {
	return func(svr *Service) *Service {
		// 这里做修改 修改完后返回指针参数
		svr.host = host
		return svr
	}
}

func WithPort(port int64) Option {
	return func(svr *Service) *Service {
		svr.port = port
		return svr
	}
}

func WithTimeout(timeout time.Duration) Option {
	return func(svr *Service) *Service {
		svr.timeout = timeout
		return svr
	}
}

func WithMaxConn(maxConn int64) Option {
	return func(svr *Service) *Service {
		svr.maxConn = maxConn
		return svr
	}
}
```

使用方式
```go
package main

import (
	"fmt"
	"time"
	server "tmp/service"
)

func main() {
	// 初始化时传多个选项参数进去
	server := server.NewService(
		server.WithHost("127.0.0.1"),
		server.WithPort(8888),
		server.WithTimeout(10*time.Minute),
		server.WithMaxConn(100),
	)

	fmt.Printf("server: %#v start \n", server)
}
```
