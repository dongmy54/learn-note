### MapReduce
循环处理收集结果，有返回值

```go
users := []User{
		{ID: 12, Name: "张三"},
		{ID: 13, Name: "李四"},
		{ID: 14, Name: "王五"},
	}

	result, err := mr.MapReduce(func(source chan<- User) {
		// 第一个函数作用：数据源
		// 参数：通道单向入参
		for _, user := range users {
			source <- user
		}
	},
		// 第二个函数 做数据的中间处理 处理完后可以进入下一个环节
		// 参数1: 上一个环节的输入类型
		// 参数2: 写入下一环节数据类型 mr.Writer[xx] xx这里指定写入数据类型
		// 参数3: 取消函数（照着抄下来就行）
		func(user User, writer mr.Writer[User], cancel func(err error)) {
			if user.ID%2 == 0 {
				writer.Write(user) // 能整除则进入下一环节
			}
		},
		// 第三个函数：最终结果收集
		// 参数1: 通道输出
		// 参数2: 写入数据的类型 mr.Writer[xx] xx这里定义的是 最终返回的类型
		// 参数3: 同上
		func(pipe <-chan User, writer mr.Writer[int], cancel func(error)) {
			// 临时收集数据
			var count int
			for u := range pipe {
				// 这里做收集
				count += u.ID
			}

			// 收集完写入 最终结果做为返回
			// PS: 只能写入一次
			writer.Write(count)
		}, mr.WithWorkers(100)) // 指定work数

	if err == nil {
		fmt.Println("============", result)
	} else {
		fmt.Println("===========发生错误==\n", err)
	}
```
